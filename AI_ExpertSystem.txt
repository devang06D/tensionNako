#include <iostream>
using namespace std;

// Inference Engine
string inferDecision(float price, string trend, int risk) {
    if (trend == "up" && risk == 3)
        return "BUY (Strong upward trend)";
    else if (trend == "down" && risk == 3)
        return "SELL (Market falling)";
    else if (trend == "stable" && price < 100)
        return "BUY (Undervalued stock)";
    else if (trend == "stable" && price > 200)
        return "SELL (Overvalued stock)";
    else
        return "HOLD (No clear signal)";
}

int main() {
    float price;
    string trend;
    int risk;

    cout << "===== Stock Market Expert System =====\n";

    // User Interface
    cout << "Enter stock price: ";
    cin >> price;

    cout << "Enter trend (up/down/stable): ";
    cin >> trend;

    cout << "Enter risk level (1-Low, 2-Medium, 3-High): ";
    cin >> risk;

    // Inference Engine call
    string result = inferDecision(price, trend, risk);

    // Output
    cout << "\nExpert Decision: " << result << endl;

    return 0;
}
