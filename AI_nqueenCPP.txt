#include <iostream>

using namespace std;

int N;

// Function to check whether queen can be placed
bool isSafe(int board[10][10], int row, int col) {

    // Check upper column
    for (int i = 0; i < row; i++) {
        if (board[i][col] == 1) {
            return false;
        }
    }

    // Check left diagonal
    for (int i = row, j = col; i >= 0 && j >= 0; i--, j--) {
        if (board[i][j] == 1) {
            return false;
        }
    }

    // Check right diagonal
    for (int i = row, j = col; i >= 0 && j < N; i--, j++) {
        if (board[i][j] == 1) {
            return false;
        }
    }

    return true;
}

// Function to solve N Queen
bool solve(int board[10][10], int row) {

    // If all queens are placed
    if (row == N) {
        return true;
    }

    // Try placing queen in every column
    for (int col = 0; col < N; col++) {

        if (isSafe(board, row, col)) {

            // Place queen
            board[row][col] = 1;

            // Recursive call
            if (solve(board, row + 1)) {
                return true;
            }

            // Backtracking
            board[row][col] = 0;
        }
    }

    return false;
}

// Print board
void printBoard(int board[10][10]) {

    cout << "\nSolution:\n";

    for (int i = 0; i < N; i++) {

        for (int j = 0; j < N; j++) {
            cout << board[i][j] << " ";
        }

        cout << endl;
    }
}

int main() {

    int board[10][10] = {0};

    cout << "Enter value of N: ";
    cin >> N;

    if (solve(board, 0)) {
        printBoard(board);
    }
    else {
        cout << "No Solution";
    }

    return 0;
}
